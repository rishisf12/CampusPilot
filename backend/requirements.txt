# --- Observability: metrics --------------------------------------------------
# prometheus_client is the reference Python client, Apache-2.0. It gives us
# /metrics, which is what Prometheus scrapes - the "DB performance" and "slow
# endpoints" numbers in the Health subsection come out of these instruments.
prometheus-client

# --- Observability: scan queue ----------------------------------------------
# RQ is BSD-3-Clause and deliberately small (~2k lines). A scan is a linear,
# expensive, on-request job; BullMQ would mean running a Node service purely as a
# queue and Celery would bring a result backend and beat scheduler we never use.
# Concurrency is pinned to 1 so scans cannot overload a small VPS.
rq
redis

# --- Feedback NLP (all local, no model downloads at runtime) ----------------
# VADER is a lexicon-based sentiment model from NLTK (Apache-2.0). It runs in
# microseconds with no torch dependency, which matters because sentiment runs
# over every feedback row on demand rather than in a batch.
vaderSentiment

# --- Database: PostgreSQL ---------------------------------------------------
# psycopg 3 (not psycopg2): SQLAlchemy 2.x supports it natively via the
# "postgresql+psycopg" driver name, and it is the actively developed line.
# [binary] ships the libpq wheel, so no libpq-dev at build time - which is why
# the Dockerfile keeps libpq-dev only as a belt-and-braces fallback.
psycopg[binary]
# Schema migrations. Replaces the ad-hoc ALTER TABLE pass that only ever worked
